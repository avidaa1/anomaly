function levenshtein(a,b){
  var m=a.length,n=b.length,d=[];
  for(var i=0;i<=m;i++) d[i]=[i];
  for(var j=0;j<=n;j++) d[0][j]=j;
  for(i=1;i<=m;i++) for(j=1;j<=n;j++){
    d[i][j] = a[i-1]===b[j-1] ? d[i-1][j-1] : Math.min(d[i-1][j-1]+1,d[i][j-1]+1,d[i-1][j]+1);
  }
  return d[m][n];
}
function fuzzyIncludes(text, key){
  if (text.indexOf(key)!==-1) return true;
  if (key.length < 5) return false;
  var words = text.split(/\s+/);
  for (var i=0;i<words.length;i++){
    if (words[i].length>=5 && levenshtein(words[i], key) <= 1) return true;
  }
  return false;
}

  });
  return best;
}

async function decryptBlob(passphrase, blob){
  var parts = blob.split('.');
  var salt = b64ToBytes(parts[1]), iv = b64ToBytes(parts[2]), ct = b64ToBytes(parts[3]);
  var enc = new TextEncoder();
  var baseKey = await crypto.subtle.importKey('raw', enc.encode(normalizeKey(passphrase)), 'PBKDF2', false, ['deriveKey']);
  var key = await crypto.subtle.deriveKey(
    { name:'PBKDF2', salt:salt, iterations:100000, hash:'SHA-256' },
    baseKey, { name:'AES-GCM', length:256 }, false, ['decrypt']
  );
  var pt = await crypto.subtle.decrypt({ name:'AES-GCM', iv:iv }, key, ct);
  return new TextDecoder().decode(pt);
}
